\newtcolorbox{plainbox}{enhanced, arc=4pt, boxrule=0pt, frame hidden, colback=mist,

\newenvironment{darkframe}
  {\setbeamercolor{background canvas}{bg=ink}\setbeamercolor{normal text}{fg=white}%
   \usebeamercolor[fg]{normal text}\begin{frame}[plain]}
  {\end{frame}}
